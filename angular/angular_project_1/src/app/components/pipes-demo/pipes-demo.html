Let's learn Angular Pipes

lowercase , uppercase, titlecase pipe
{{userName}}
{{userName | uppercase}}
{{userName | lowercase}}
{{userName | titlecase}}
currency pipe
{{salary}}
{{salary | currency}}
{{salary | currency : 'INR'}}
{{salary | currency : '€'}}
{{salary | currency : '¥'}}
{{salary | currency : '¥'}}
{{dateObj}}
{{dateObj | date}}
{{dateObj | date : 'short'}}
{{dateObj | date : 'medium'}}
{{dateObj | date : 'long'}}
{{dateObj | date : 'full'}}
{{dateObj | date : 'shortDate'}}
{{dateObj | date : 'mediumDate'}}
{{dateObj | date : 'longDate'}}
{{dateObj | date : 'fullDate'}}
{{dateObj | date : 'shortTime'}}
{{dateObj | date : 'mediumTime'}}
{{dateObj | date : 'longTime'}}
{{dateObj | date : 'fullTime'}}
{{ dateObj | date : 'dd-MMM-yyyy' }}
{{ dateObj | date : 'hh:mm:ss a' }}
{{ dateObj | date: "fullTime":"UTC" }}
{{ dateObj | date: "fullTime":"IST" }}
{{ dateObj | date: "EEEE" }}
number & percent pipe
{{100000000 | number}}
{{ 12.111222333 | number: "3.2-5" }}
{{ 12.1 | number: "3.2-5" }}
{{ 0.5 | number:'3.2-5'}}
{{ 2.5 | percent}}
{{ 2.5 | percent:'2.2-5'}}
{{0.024 | percent}}
json and keyvalue pipe:
{{user}}
{{user | json}}

}
slice pipe can be used with array and string
{{cars}}
{{cars | slice : 2 : 4}}
{{'Angular is a Framework' | slice : 0 : 7}}
{{'Angular is a Framework' | slice : 0 : -4}}
